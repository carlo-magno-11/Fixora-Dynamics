# Fixora foundation migration — reviewed SQL draft

This is a reviewed implementation draft for the migration generated by the Supabase CLI. It is not a migration file and must not be applied remotely. Create the migration with `supabase migration new fixora_identity_and_businesses`, then copy the SQL into the generated file in Codespaces.

The SQL intentionally uses Fixora-prefixed public tables. It does not move, rename, drop, or replay any EYESITE object.

## Migration SQL

```sql
create schema if not exists private;
revoke all on schema private from public;
revoke all on schema private from anon;
revoke all on schema private from authenticated;
grant usage on schema private to authenticated;

create table public.fixora_profiles (
  id uuid primary key references auth.users(id) on delete cascade,
  email text,
  display_name text,
  locale text not null default 'es-MX'
    check (locale in ('es-MX', 'en-US')),
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

create table public.fixora_businesses (
  id uuid primary key default gen_random_uuid(),
  name text not null
    check (char_length(btrim(name)) between 2 and 100),
  business_type text not null default 'general'
    check (business_type in (
      'general',
      'retail',
      'food',
      'services_beauty',
      'digital_products',
      'workshop'
    )),
  currency_code text not null default 'MXN'
    check (currency_code in ('MXN', 'USD')),
  timezone text not null default 'America/Merida',
  created_by uuid not null references auth.users(id),
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

create table public.fixora_business_memberships (
  business_id uuid not null
    references public.fixora_businesses(id) on delete cascade,
  user_id uuid not null references auth.users(id) on delete cascade,
  role text not null
    check (role in ('owner', 'admin', 'editor', 'viewer')),
  status text not null default 'active'
    check (status in ('active', 'invited', 'suspended')),
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now(),
  primary key (business_id, user_id)
);

create index fixora_business_memberships_user_status_idx
  on public.fixora_business_memberships (user_id, status);

create index fixora_business_memberships_business_status_idx
  on public.fixora_business_memberships (business_id, status);

create or replace function private.set_updated_at()
returns trigger
language plpgsql
set search_path = ''
as $$
begin
  new.updated_at := pg_catalog.now();
  return new;
end;
$$;

revoke all on function private.set_updated_at() from public, anon, authenticated;

create trigger fixora_profiles_set_updated_at
before update on public.fixora_profiles
for each row execute function private.set_updated_at();

create trigger fixora_businesses_set_updated_at
before update on public.fixora_businesses
for each row execute function private.set_updated_at();

create trigger fixora_memberships_set_updated_at
before update on public.fixora_business_memberships
for each row execute function private.set_updated_at();

create or replace function private.has_active_business_role(
  p_business_id uuid,
  p_allowed_roles text[] default null
)
returns boolean
language sql
stable
security definer
set search_path = ''
as $$
  select exists (
    select 1
    from public.fixora_business_memberships as membership
    where membership.business_id = p_business_id
      and membership.user_id = (select auth.uid())
      and membership.status = 'active'
      and (
        p_allowed_roles is null
        or membership.role = any (p_allowed_roles)
      )
  );
$$;

revoke all on function private.has_active_business_role(uuid, text[]) from public, anon;
grant execute on function private.has_active_business_role(uuid, text[]) to authenticated;

create or replace function private.handle_new_auth_user()
returns trigger
language plpgsql
security definer
set search_path = ''
as $$
declare
  v_display_name text;
begin
  v_display_name :=
    coalesce(
      nullif(pg_catalog.btrim(new.raw_user_meta_data ->> 'display_name'), ''),
      nullif(pg_catalog.split_part(coalesce(new.email, ''), '@', 1), '')
    );

  insert into public.fixora_profiles (id, email, display_name)
  values (new.id, new.email, v_display_name)
  on conflict (id) do update
    set email = excluded.email,
        updated_at = pg_catalog.now();

  return new;
end;
$$;

revoke all on function private.handle_new_auth_user() from public, anon, authenticated;

create trigger fixora_on_auth_user_created
after insert on auth.users
for each row execute function private.handle_new_auth_user();

create or replace function public.fixora_create_business(
  p_name text,
  p_business_type text default 'general',
  p_currency_code text default 'MXN',
  p_timezone text default 'America/Merida'
)
returns uuid
language plpgsql
security definer
set search_path = ''
as $$
declare
  v_user_id uuid;
  v_business_id uuid;
  v_name text;
  v_timezone text;
begin
  v_user_id := auth.uid();

  if v_user_id is null then
    raise exception 'Authentication required'
      using errcode = '28000';
  end if;

  v_name := pg_catalog.btrim(coalesce(p_name, ''));

  if pg_catalog.char_length(v_name) < 2
     or pg_catalog.char_length(v_name) > 100 then
    raise exception 'Business name must contain 2 to 100 characters'
      using errcode = '22023';
  end if;

  if p_business_type is null or p_business_type not in (
    'general',
    'retail',
    'food',
    'services_beauty',
    'digital_products',
    'workshop'
  ) then
    raise exception 'Unsupported business type'
      using errcode = '22023';
  end if;

  if p_currency_code is null or p_currency_code not in ('MXN', 'USD') then
    raise exception 'Unsupported currency'
      using errcode = '22023';
  end if;

  v_timezone := coalesce(nullif(pg_catalog.btrim(p_timezone), ''), 'America/Merida');

  if not exists (
    select 1
    from pg_catalog.pg_timezone_names as timezone_name
    where timezone_name.name = v_timezone
  ) then
    raise exception 'Unsupported timezone'
      using errcode = '22023';
  end if;

  insert into public.fixora_businesses (
    name,
    business_type,
    currency_code,
    timezone,
    created_by
  )
  values (
    v_name,
    p_business_type,
    p_currency_code,
    v_timezone,
    v_user_id
  )
  returning id into v_business_id;

  insert into public.fixora_business_memberships (
    business_id,
    user_id,
    role,
    status
  )
  values (
    v_business_id,
    v_user_id,
    'owner',
    'active'
  );

  return v_business_id;
end;
$$;

revoke all on function public.fixora_create_business(text, text, text, text)
  from public, anon;
grant execute on function public.fixora_create_business(text, text, text, text)
  to authenticated;

alter table public.fixora_profiles enable row level security;
alter table public.fixora_businesses enable row level security;
alter table public.fixora_business_memberships enable row level security;

create policy "fixora_profiles_select_self"
on public.fixora_profiles
for select
to authenticated
using ((select auth.uid()) = id);

create policy "fixora_profiles_update_self"
on public.fixora_profiles
for update
to authenticated
using ((select auth.uid()) = id)
with check ((select auth.uid()) = id);

create policy "fixora_businesses_select_active_member"
on public.fixora_businesses
for select
to authenticated
using (
  (select private.has_active_business_role(id, null))
);

create policy "fixora_businesses_update_owner_admin"
on public.fixora_businesses
for update
to authenticated
using (
  (select private.has_active_business_role(id, array['owner', 'admin']::text[]))
)
with check (
  (select private.has_active_business_role(id, array['owner', 'admin']::text[]))
);

create policy "fixora_memberships_select_active_member"
on public.fixora_business_memberships
for select
to authenticated
using (
  (select private.has_active_business_role(business_id, null))
);

revoke all on table public.fixora_profiles
  from public, anon, authenticated;
revoke all on table public.fixora_businesses
  from public, anon, authenticated;
revoke all on table public.fixora_business_memberships
  from public, anon, authenticated;

grant select on table public.fixora_profiles to authenticated;
grant update (display_name, locale) on table public.fixora_profiles to authenticated;

grant select on table public.fixora_businesses to authenticated;
grant update (name, business_type, currency_code, timezone)
  on table public.fixora_businesses to authenticated;

grant select on table public.fixora_business_memberships to authenticated;
```

## Important review notes

- The signup trigger is safe for the isolated local database. The remote project still has a legacy signup trigger. Do not enable remote signup until a separate reviewed migration removes/replaces the legacy trigger and verifies the resulting trigger set.
- The profile trigger only copies a display label from user metadata. It does not treat metadata as a role or permission.
- The private membership helper is SECURITY DEFINER to avoid membership-policy recursion. It is not in an exposed API schema, has a fixed empty search_path, fully qualified references, and only authenticated execution.
- The business RPC derives the owner from `auth.uid()`; it does not accept a user ID or role from the client.
- The first migration deliberately has no delete policy, no direct membership mutation grants, and no client insert grant for businesses.
- Do not add `FORCE ROW LEVEL SECURITY` without testing the migration owner and Auth trigger behavior.
- Check the generated SQL against the installed local Supabase/PostgreSQL runtime before accepting it.

## Local validation checklist

After creating the migration in Codespaces:

1. Inspect the generated migration filename and the complete SQL.
2. Run a clean local database reset using the local-only command confirmed by `supabase db reset --help`.
3. Confirm exactly these three Fixora tables exist and all three have RLS enabled.
4. Confirm `private` is not in exposed API schemas.
5. Confirm anon has no table privileges and authenticated has only the listed grants.
6. Confirm the Auth trigger creates one `fixora_profiles` row and no EYESITE table is referenced by the migration.
7. Test `fixora_create_business` as authenticated and confirm business + owner membership are atomic.
8. Test the RPC as anon and confirm it fails.
9. Test two users and two businesses: cross-tenant SELECT/UPDATE must reveal or change zero rows.
10. Confirm a viewer cannot update a business, and no client can mutate memberships.
11. Run typecheck, unit tests, build, and Supabase advisors.

No remote deployment is part of this checklist.
